* Flake
This configuration is flake based.
#+begin_src nix :tangle flake.nix :noweb yes
  {
    description = "Home Manager configuration of douwe";

    inputs = {
      nixpkgs.url = "github:nixos/nixpkgs/nixpkgs-unstable";
      <<flake-inputs>>
    };

    outputs =
      { nixpkgs, nixos-hardware, home-manager, ... }:
      let
        system = "x86_64-linux";
        lib = nixpkgs.lib;
        pkgs = nixpkgs.legacyPackages.${system};
      in
      {
        nixosConfigurations = {
          <<nixos-configs>>
        };
        homeConfigurations = {
          <<home-configs>>
        };
      };
  }
#+end_src

** NixOS setup
I load NixOS hardware for laptop specific configuration, so we add this to the flake inputs.
#+begin_src nix :noweb-ref flake-inputs
  nixos-hardware = {
    url = "github:NixOS/nixos-hardware";
    inputs.nixpkgs.follows = "nixpkgs";
  };
#+end_src

Declare my laptop as a NixOS system.
#+begin_src nix :noweb-ref nixos-configs
  dlaptop = lib.nixosSystem {
    system = "x86_64-linux";
    modules = [ 
      nixos-hardware.nixosModules.lenovo-ideapad-16ach6
      ./configuration.nix 
    ];
  };
#+end_src

** Home manager
For my personal programs and settings, I use home-manager, so we also
add this to the inputs.

#+begin_src nix :noweb-ref flake-inputs
  home-manager = {
    url = "github:nix-community/home-manager";
    inputs.nixpkgs.follows = "nixpkgs";
  };
#+end_src

And we declare a home profile for me.
#+begin_src nix :noweb-ref home-configs
  "douwe" = home-manager.lib.homeManagerConfiguration {
    inherit pkgs;
    
    modules = [ ./home.nix ];
  };
#+end_src
